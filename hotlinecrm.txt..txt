        const leadRef = doc(state.db, "artifacts", state.appId, "public", "data", "crm_leads", docId);
        await deleteDoc(leadRef);
      } else {
        state.leads = state.leads.filter((l) => l.id !== docId);
        renderAll();
      }
    }

    async function saveBuyerToStore(docId, buyerData) {
      if (state.cloudReady && state.db && state.user) {
        const buyersColRef = collection(state.db, "artifacts", state.appId, "public", "data", "crm_buyers");
        if (docId) {
          await updateDoc(doc(buyersColRef, docId), buyerData);
        } else {
          await addDoc(buyersColRef, { ...buyerData, order: state.buyers.length + 1 });
        }
      } else {
        if (docId) {
          const idx = state.buyers.findIndex((b) => b.id === docId);
          if (idx !== -1) state.buyers[idx] = { ...state.buyers[idx], ...buyerData };
        } else {
          state.buyers.push({ id: "buyer-" + Date.now(), ...buyerData, order: state.buyers.length + 1 });
        }
        renderAll();
      }
    }

    async function deleteBuyerFromStore(docId) {
      if (state.cloudReady && state.db && state.user) {
        const buyerRef = doc(state.db, "artifacts", state.appId, "public", "data", "crm_buyers", docId);
        await deleteDoc(buyerRef);
      } else {
        state.buyers = state.buyers.filter((b) => b.id !== docId);
        renderAll();
      }
    }

    async function saveCredentialsToStore(patch) {
      const updated = { ...state.credentials, ...patch };
      if (state.cloudReady && state.db && state.user) {
        const configDocRef = doc(state.db, "artifacts", state.appId, "public", "data", "crm_config", "main");
        await setDoc(configDocRef, updated, { merge: true });
      } else {
        state.credentials = updated;
        populateSettingsInputs();
      }
    }

    // --- Per-Buyer Phone Duplicate Check ---
    function findDuplicateForBuyer(phoneRaw, buyerId, ignoreLeadDocId = null) {
      const normalized = normalizePhone(phoneRaw);
      if (normalized.length < 7 || !buyerId) return null;
      return (
        state.leads.find(
          (l) =>
            l.id !== ignoreLeadDocId &&
            l.buyerId === buyerId &&
            normalizePhone(l.phone) === normalized
        ) || null
      );
    }

    function updateLiveDuplicateFeedback() {
      const phoneVal = document.getElementById("lead-phone")?.value || "";
      const buyerId = document.getElementById("lead-buyer-select")?.value || "";
      const feedback = document.getElementById("buyer-dup-feedback");
      if (!feedback) return;

      const norm = normalizePhone(phoneVal);
      if (norm.length < 7) {
        feedback.className = "text-xs mt-1.5 text-slate-500";
        feedback.textContent = "Checks phone number uniqueness strictly for the selected buyer.";
        return;
      }

      const buyerObj = state.buyers.find((b) => b.id === buyerId);
      const buyerName = buyerObj ? buyerObj.name : "selected buyer";
      const dup = findDuplicateForBuyer(phoneVal, buyerId);

      if (dup) {
        feedback.className = "text-xs mt-1.5 text-rose-600 font-semibold";
        feedback.innerHTML = `<i class="fa-solid fa-triangle-exclamation mr-1"></i>Duplicate blocked: ${escapeHtml(phoneVal)} was already sent to ${escapeHtml(buyerName)} (${escapeHtml(dup.leadCode)}). Select another buyer.`;
      } else {
        feedback.className = "text-xs mt-1.5 text-emerald-700 font-semibold";
        feedback.innerHTML = `<i class="fa-solid fa-circle-check mr-1"></i>Clear for ${escapeHtml(buyerName)} — no prior submission on this buyer.`;
      }
    }

    function renderBuyerDropdowns() {
      const activeBuyers = state.buyers.filter((b) => b.status === "Active");

      // Create Lead Buyer Select (Active buyers only)
      const intakeSelect = document.getElementById("lead-buyer-select");
      if (intakeSelect) {
        const prev = intakeSelect.value;
        if (activeBuyers.length === 0) {
          intakeSelect.innerHTML = `<option value="">No Active Buyers Available</option>`;
        } else {
          intakeSelect.innerHTML = activeBuyers
            .map((b) => `<option value="${escapeHtml(b.id)}">${escapeHtml(b.name)}</option>`)
            .join("");
          if (activeBuyers.some((b) => b.id === prev)) {
            intakeSelect.value = prev;
          }
        }
      }

      // Edit Modal Buyer Select (All buyers)
      const editSelect = document.getElementById("edit-buyer-id");
      if (editSelect) {
        editSelect.innerHTML = state.buyers
          .map((b) => `<option value="${escapeHtml(b.id)}">${escapeHtml(b.name)} (${escapeHtml(b.status)})</option>`)
          .join("");
      }

      // Queue Filter Buyer Select
      const filterSelect = document.getElementById("queue-buyer-filter");
      if (filterSelect) {
        const prev = filterSelect.value || "ALL";
        filterSelect.innerHTML =
          `<option value="ALL">All Buyers</option>` +
          state.buyers
            .map((b) => `<option value="${escapeHtml(b.id)}">${escapeHtml(b.name)}</option>`)
            .join("");
        filterSelect.value = prev;
      }

      // Queue Filter Agent Select
      const agentFilter = document.getElementById("queue-agent-filter");
      if (agentFilter) {
        const prev = agentFilter.value || "ALL";
        const uniqueAgents = [...new Set(state.leads.map((l) => l.agentName).filter(Boolean))].sort();
        agentFilter.innerHTML =
          `<option value="ALL">All Agents</option>` +
          uniqueAgents.map((a) => `<option value="${escapeHtml(a)}">${escapeHtml(a)}</option>`).join("");
        agentFilter.value = prev;
      }
    }

    function renderDashboard() {

          { label: "Pending Leads", count: pending, color: "bg-amber-500" },
          { label: "Rejected Leads", count: rejected, color: "bg-rose-500" }

          })
          .join("");

    function renderLeadsQueue() {
      const tbody = document.getElementById("leads-queue-tbody");
      if (!tbody) return;

      const statusFilterMap = {
        "all-leads": "ALL",
        "quoted-leads": "Quoted",
        "rejected-leads": "Rejected",
        "pending-leads": "Pending"
      };

      const requiredStatus = statusFilterMap[state.activeTab] || "ALL";
      const query = (document.getElementById("queue-search-input")?.value || "").trim().toLowerCase();
      const buyerFilter = document.getElementById("queue-buyer-filter")?.value || "ALL";
      const agentFilter = document.getElementById("queue-agent-filter")?.value || "ALL";

      const filtered = state.leads.filter((l) => {
        if (requiredStatus !== "ALL" && l.status !== requiredStatus) return false;
        if (buyerFilter !== "ALL" && l.buyerId !== buyerFilter) return false;
        if (agentFilter !== "ALL" && l.agentName !== agentFilter) return false;
        if (query) {
          const haystack = [
            l.leadCode,
            l.firstName,
            l.lastName,
            l.phone,
            l.city,
            l.state,
            l.zip,
            l.agentName,
            l.makeModels,
            l.comment,
            getBuyerName(l.buyerId)
          ]
            .join(" ")
            .toLowerCase();
          if (!haystack.includes(query)) return false;
        }
        return true;
      });

      if (filtered.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="8" class="py-10 px-4 text-center text-slate-500">
              No leads match the current view or filter criteria.
            </td>
          </tr>
        `;
        return;
      }

      const isAdmin = state.sessionRole === "admin";

      tbody.innerHTML = filtered
        .map((lead) => {
          const badgeClass =
            lead.status === "Quoted"
              ? "bg-emerald-50 text-emerald-700 border-emerald-200"
              : lead.status === "Rejected"
              ? "bg-rose-50 text-rose-700 border-rose-200"
              : "bg-amber-50 text-amber-700 border-amber-200";

          return `
            <tr class="hover:bg-blue-50/30 transition">
              <td class="py-3.5 px-4 align-top">
                <div class="font-mono font-bold text-blue-700 text-xs">${escapeHtml(lead.leadCode || "HB-LEAD")}</div>

              </td>
              <td class="py-3.5 px-4 align-top">
                <div class="font-bold text-slate-900">${escapeHtml(lead.firstName)} ${escapeHtml(lead.lastName)}</div>
                <div class="text-xs text-slate-500 mt-0.5">${escapeHtml(lead.vehicles)} veh • ${escapeHtml(lead.makeModels)}</div>
                <div class="text-[11px] text-slate-400">Ins: ${escapeHtml(lead.insurance)}</div>
              </td>
              <td class="py-3.5 px-4 align-top">
                <div class="font-mono font-semibold text-slate-800">${escapeHtml(lead.phone)}</div>
                <div class="text-xs text-slate-500 mt-0.5">${escapeHtml(lead.city)}, ${escapeHtml(lead.state)} ${escapeHtml(lead.zip)}</div>
              </td>
              <td class="py-3.5 px-4 align-top">
                <span class="px-2.5 py-1 rounded-lg bg-blue-50 text-blue-800 text-xs font-semibold">
                  ${escapeHtml(getBuyerName(lead.buyerId))}

                ${escapeHtml(lead.agentName)}

    function switchTab(tabId) {
      const isAdmin = state.sessionRole === "admin";
      if (!isAdmin && (tabId === "buyers-section" || tabId === "users-settings")) {
        showToast("Administrator access required for that section.", true);
        return;
      }

      state.activeTab = tabId;

      // Update sidebar button styles
      document.querySelectorAll(".nav-tab-btn").forEach((btn) => {
        const active = btn.getAttribute("data-tab") === tabId;
        btn.className = active
          ? "nav-tab-btn w-full flex items-center justify-between px-3.5 py-2.5 rounded-xl text-xs sm:text-sm font-semibold bg-blue-600 text-white shadow-sm transition cursor-pointer"
          : "nav-tab-btn w-full flex items-center justify-between px-3.5 py-2.5 rounded-xl text-xs sm:text-sm font-medium text-blue-100 hover:bg-blue-800/60 transition cursor-pointer";
      });

      // Toggle views
      document.querySelectorAll(".crm-view").forEach((v) => v.classList.add("hidden"));

      if (["all-leads", "quoted-leads", "rejected-leads", "pending-leads"].includes(tabId)) {
        document.getElementById("view-leads-queue").classList.remove("hidden");
        renderLeadsQueue();
      } else {
        const target = document.getElementById("view-" + tabId);
        if (target) target.classList.remove("hidden");
      }

      const meta = TAB_META[tabId] || { title: "HOTLINE BPO", subtitle: "" };
      document.getElementById("topbar-title").textContent = meta.title;
      document.getElementById("topbar-subtitle").textContent = meta.subtitle;

      // Close mobile sidebar if open
      document.getElementById("crm-sidebar").classList.add("-translate-x-full");
      document.getElementById("sidebar-backdrop").classList.add("hidden");
    }

      document.getElementById("edit-address").value = lead.address || "";
      document.getElementById("edit-city").value = lead.city || "";
      document.getElementById("edit-state").value = lead.state || "";
      document.getElementById("edit-zip").value = lead.zip || "";
      document.getElementById("edit-phone").value = lead.phone || "";
      document.getElementById("edit-vehicles").value = lead.vehicles || 1;
      document.getElementById("edit-make-models").value = lead.makeModels || "";
      document.getElementById("edit-housing").value = lead.housing || "Homeowner";
      document.getElementById("edit-insurance").value = lead.insurance || "";
      document.getElementById("edit-agent-name").value = lead.agentName || "";
      document.getElementById("edit-buyer-id").value = lead.buyerId || "";
      document.getElementById("edit-status").value = lead.status || "Pending";
      document.getElementById("edit-comment").value = lead.comment || "";

      const overrideBadge = document.getElementById("edit-override-badge");
      const auditNote = document.getElementById("edit-audit-note");
      if (lead.adminOverridden) {
        overrideBadge.classList.remove("hidden");
        auditNote.textContent = lead.overrideNote || "Updated by Administrator.";
      } else {
        overrideBadge.classList.add("hidden");
        auditNote.textContent = isAdmin
          ? "As Admin, changing status or comment here updates all devices immediately."
          : "Read-only view. Contact Administrator for status or comment overrides.";
      }

      // Toggle read-only vs editable fields for Agent vs Admin
      const formElements = document.querySelectorAll("#lead-edit-form input, #lead-edit-form select, #lead-edit-form textarea");
      formElements.forEach((el) => {
        if (el.type !== "hidden") {
          el.disabled = !isAdmin;
        }
      });

      document.querySelectorAll(".admin-only-el").forEach((el) => {
        el.classList.toggle("hidden", !isAdmin);
      });

      document.getElementById("lead-modal").classList.remove("hidden");
    }

    // --- CSV Export ---
    function exportLeadsCsv() {
      const headers = [
        "Lead ID",
        "First Name",
        "Last Name",
        "DOB",
        "Address",
        "City",
        "State",
        "Zip",
        "Phone",
        "Vehicles",
        "Make & Models",
        "Housing",
        "Current Insurance",
        "Agent Name",
        "Buyer",
        "Status",
        "Comment"
      ];

      const rows = state.leads.map((l) => [
        l.leadCode,
        l.firstName,
        l.lastName,
        l.dob,
        l.address,
        l.city,
        l.state,
        l.zip,
        l.phone,
        l.vehicles,
        l.makeModels,
        l.housing,
        l.insurance,
        l.agentName,
        getBuyerName(l.buyerId),
        l.status,
        l.comment
      ]);

      const csvContent =
        [headers, ...rows]
          .map((row) => row.map((cell) => `"${String(cell ?? "").replace(/"/g, '""')}"`).join(","))
          .join("\n");

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `hotline-bpo-leads-${new Date().toISOString().slice(0, 10)}.csv`;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
      showToast("CSV report exported.");
    }

    // --- Event Listeners Setup ---
    function bindEvents() {
      // Login form
      document.getElementById("login-form").addEventListener("submit", (e) => {
        e.preventDefault();
        const email = document.getElementById("login-email").value.trim().toLowerCase();
        const pass = document.getElementById("login-password").value;
        const errBox = document.getElementById("login-error-box");

        if (!state.cloudReady || !state.user) {
          errBox.textContent =
            state.cloudError ||
            "Cloud database is not connected. Configure Firebase, enable Anonymous Auth, and refresh.";
          errBox.classList.remove("hidden");
          return;
        }

        const activeAdminEmail = (state.credentials.adminEmail || DEFAULT_CREDENTIALS.adminEmail).trim().toLowerCase();
        const activeAdminPass = state.credentials.adminPassword || DEFAULT_CREDENTIALS.adminPassword;
        const activeAgentEmail = (state.credentials.agentEmail || DEFAULT_CREDENTIALS.agentEmail).trim().toLowerCase();
        const activeAgentPass = state.credentials.agentPassword || DEFAULT_CREDENTIALS.agentPassword;

        if (email === activeAdminEmail && pass === activeAdminPass) {
          errBox.classList.add("hidden");
          applySessionRole("admin", state.credentials.adminEmail.trim());
        } else if (email === activeAgentEmail && pass === activeAgentPass) {
          errBox.classList.add("hidden");
          applySessionRole("agent", state.credentials.agentEmail.trim());
        } else {
          errBox.textContent = "Invalid email or password. Please verify your access credentials.";
          errBox.classList.remove("hidden");
        }
      });

      document.getElementById("toggle-password-visibility").addEventListener("click", () => {
        const inp = document.getElementById("login-password");
        inp.type = inp.type === "password" ? "text" : "password";
      });

      document.getElementById("logout-btn").addEventListener("click", () => {
        state.sessionRole = null;
        state.sessionEmail = null;
        document.getElementById("login-password").value = "";
        document.getElementById("crm-app").classList.add("hidden");
        document.getElementById("login-screen").classList.remove("hidden");
      });

      // Sidebar navigation
      document.querySelectorAll(".nav-tab-btn").forEach((btn) => {
        btn.addEventListener("click", () => {
          switchTab(btn.getAttribute("data-tab"));
        });
      });

      document.getElementById("quick-create-lead-btn").addEventListener("click", () => {
        switchTab("create-lead");
      });

      document.getElementById("open-sidebar-btn").addEventListener("click", () => {
        document.getElementById("crm-sidebar").classList.remove("-translate-x-full");
        document.getElementById("sidebar-backdrop").classList.remove("hidden");
      });

      document.getElementById("close-sidebar-btn").addEventListener("click", () => {
        document.getElementById("crm-sidebar").classList.add("-translate-x-full");
        document.getElementById("sidebar-backdrop").classList.add("hidden");
      });

      document.getElementById("sidebar-backdrop").addEventListener("click", () => {
        document.getElementById("crm-sidebar").classList.add("-translate-x-full");
        document.getElementById("sidebar-backdrop").classList.add("hidden");
      });

      // Mutually exclusive Homeowner / Renter checkboxes
      document.querySelectorAll(".housing-checkbox").forEach((chk) => {
        chk.addEventListener("change", () => {
          if (chk.checked) {
            document.querySelectorAll(".housing-checkbox").forEach((other) => {
              if (other !== chk) other.checked = false;
            });
          }
        });
      });

      // Live duplicate check on phone & buyer inputs
      document.getElementById("lead-phone").addEventListener("input", updateLiveDuplicateFeedback);
      document.getElementById("lead-buyer-select").addEventListener("change", updateLiveDuplicateFeedback);

      // Create Lead Form Submit
      document.getElementById("create-lead-form").addEventListener("submit", async (e) => {
        e.preventDefault();

        const homeownerChecked = document.getElementById("chk-homeowner").checked;
        const renterChecked = document.getElementById("chk-renter").checked;
        if (!homeownerChecked && !renterChecked) {
          showToast("Please check either Homeowner or Renter.", true);
          return;
        }

        const buyerId = document.getElementById("lead-buyer-select").value;
        if (!buyerId) {
          showToast("Please select an active buyer.", true);
          return;
        }

        const phone = document.getElementById("lead-phone").value.trim();
        const duplicate = findDuplicateForBuyer(phone, buyerId);
        if (duplicate) {
          showToast(
            `Duplicate blocked: ${phone} was already sent to ${getBuyerName(buyerId)} (${duplicate.leadCode}).`,
            true
          );
          return;
        }

        const nextNum = 1001 + state.leads.length;
        const newLead = {
          leadCode: `HB-${nextNum}`,
          firstName: document.getElementById("lead-first-name").value.trim(),
          lastName: document.getElementById("lead-last-name").value.trim(),
          dob: document.getElementById("lead-dob").value,
          address: document.getElementById("lead-address").value.trim(),
          zip: document.getElementById("lead-zip").value.trim(),
          city: document.getElementById("lead-city").value.trim(),
          state: document.getElementById("lead-state").value.trim().toUpperCase(),
          phone,
          vehicles: Number(document.getElementById("lead-vehicles").value) || 1,
          makeModels: document.getElementById("lead-make-models").value.trim(),
          housing: homeownerChecked ? "Homeowner" : "Renter",
          insurance: document.getElementById("lead-insurance").value.trim(),
          agentName: document.getElementById("lead-agent-name").value.trim(),
          buyerId,
          status: document.getElementById("lead-status-select").value,
          comment: document.getElementById("lead-comment").value.trim(),
          adminOverridden: false,
          overrideNote: "",
          createdAt: Date.now()
        };

        try {
          await saveLeadToStore(newLead);
          e.target.reset();
          updateLiveDuplicateFeedback();
          showToast(`Lead ${newLead.leadCode} saved and synced to cloud.`);
        } catch (err) {
          console.error(err);
          showToast("Failed to save lead to cloud.", true);
        }
      });

      // Queue filters
      document.getElementById("queue-search-input").addEventListener("input", renderLeadsQueue);
      document.getElementById("queue-buyer-filter").addEventListener("change", renderLeadsQueue);
      document.getElementById("queue-agent-filter").addEventListener("change", renderLeadsQueue);

      // Delegated actions in Lead Queue & Buyers Section
      document.addEventListener("click", (e) => {
        const openBtn = e.target.closest(".open-lead-btn");
        if (openBtn) {
          openLeadModal(openBtn.getAttribute("data-open-lead"));
          return;
        }

        const delLeadBtn = e.target.closest(".delete-lead-btn");
        if (delLeadBtn && state.sessionRole === "admin") {
          const docId = delLeadBtn.getAttribute("data-delete-lead");
          const lead = state.leads.find((l) => l.id === docId);
          openConfirmModal(
            "Delete Lead Record",
            `Are you sure you want to permanently delete ${lead ? lead.leadCode : "this lead"} across all devices?`,
            async () => {
              await deleteLeadFromStore(docId);
              showToast("Lead deleted across all devices.");
            }
          );
          return;
        }

        const editBuyerBtn = e.target.closest(".edit-buyer-btn");
        if (editBuyerBtn && state.sessionRole === "admin") {
          const docId = editBuyerBtn.getAttribute("data-edit-buyer");
          const buyer = state.buyers.find((b) => b.id === docId);
          if (buyer) {
            document.getElementById("buyer-form-heading").textContent = `Edit ${buyer.name}`;
            document.getElementById("buyer-edit-doc-id").value = buyer.id;
            document.getElementById("buyer-name-input").value = buyer.name;
            document.getElementById("buyer-cap-input").value = buyer.dailyCap;
            document.getElementById("buyer-status-input").value = buyer.status;
            document.getElementById("cancel-buyer-edit-btn").classList.remove("hidden");
          }
          return;
        }

        const delBuyerBtn = e.target.closest(".delete-buyer-btn");
        if (delBuyerBtn && state.sessionRole === "admin") {
          const docId = delBuyerBtn.getAttribute("data-delete-buyer");
          const buyer = state.buyers.find((b) => b.id === docId);
          openConfirmModal(
            "Remove Buyer",
            `Remove "${buyer ? buyer.name : "this buyer"}" from the routing list across all devices?`,
            async () => {
              await deleteBuyerFromStore(docId);
              showToast("Buyer removed across all devices.");
            }
          );
        }
      });

      // Lead Modal Save / Delete / Close
      document.getElementById("close-lead-modal-btn").addEventListener("click", closeLeadModal);
      document.getElementById("cancel-lead-modal-btn").addEventListener("click", closeLeadModal);

      document.getElementById("modal-delete-lead-btn").addEventListener("click", () => {
        if (state.sessionRole !== "admin") return;
        const docId = document.getElementById("edit-lead-doc-id").value;
        const lead = state.leads.find((l) => l.id === docId);
        openConfirmModal(
          "Delete Lead Record",
          `Permanently delete ${lead ? lead.leadCode : "this lead"} across all devices?`,
          async () => {
            await deleteLeadFromStore(docId);
            closeLeadModal();
            showToast("Lead deleted across all devices.");
          }
        );
      });

      document.getElementById("lead-edit-form").addEventListener("submit", async (e) => {
        e.preventDefault();
        if (state.sessionRole !== "admin") return;

        const docId = document.getElementById("edit-lead-doc-id").value;
        const existing = state.leads.find((l) => l.id === docId);
        if (!existing) return;

        const newBuyerId = document.getElementById("edit-buyer-id").value;
        const newPhone = document.getElementById("edit-phone").value.trim();
        const dup = findDuplicateForBuyer(newPhone, newBuyerId, docId);
        if (dup) {
          showToast(`Duplicate blocked: ${newPhone} already exists on ${getBuyerName(newBuyerId)}.`, true);
          return;
        }

        const newStatus = document.getElementById("edit-status").value;
        const newComment = document.getElementById("edit-comment").value.trim();
        const statusOrCommentChanged = existing.status !== newStatus || existing.comment !== newComment;

        await updateLeadInStore(docId, {
          firstName: document.getElementById("edit-first-name").value.trim(),
          lastName: document.getElementById("edit-last-name").value.trim(),
          dob: document.getElementById("edit-dob").value,
          address: document.getElementById("edit-address").value.trim(),
          city: document.getElementById("edit-city").value.trim(),
          state: document.getElementById("edit-state").value.trim().toUpperCase(),
          zip: document.getElementById("edit-zip").value.trim(),
          phone: newPhone,
          vehicles: Number(document.getElementById("edit-vehicles").value) || 1,
          makeModels: document.getElementById("edit-make-models").value.trim(),
          housing: document.getElementById("edit-housing").value,
          insurance: document.getElementById("edit-insurance").value.trim(),
          agentName: document.getElementById("edit-agent-name").value.trim(),
          buyerId: newBuyerId,
          status: newStatus,
          comment: newComment,
          adminOverridden: existing.adminOverridden || statusOrCommentChanged,
          overrideNote: statusOrCommentChanged
            ? `Admin override (${newStatus}) on ${new Date().toLocaleString()}`
            : existing.overrideNote || ""
        });

        closeLeadModal();
        showToast("Lead updated and synced across all devices.");
      });

      // Confirm Modal buttons
      document.getElementById("confirm-cancel-btn").addEventListener("click", closeConfirmModal);
      document.getElementById("confirm-ok-btn").addEventListener("click", async () => {
        const cb = state.pendingConfirmCallback;
        closeConfirmModal();
        if (typeof cb === "function") await cb();
      });

      // Buyer Form Submit & Cancel
      document.getElementById("buyer-form").addEventListener("submit", async (e) => {
        e.preventDefault();
        if (state.sessionRole !== "admin") return;

        const docId = document.getElementById("buyer-edit-doc-id").value;
        const name = document.getElementById("buyer-name-input").value.trim();
        const dailyCap = Number(document.getElementById("buyer-cap-input").value) || 50;
        const status = document.getElementById("buyer-status-input").value;

        await saveBuyerToStore(docId || null, { name, dailyCap, status });

        document.getElementById("buyer-form").reset();
        document.getElementById("buyer-edit-doc-id").value = "";
        document.getElementById("buyer-form-heading").textContent = "Add New Buyer";
        document.getElementById("cancel-buyer-edit-btn").classList.add("hidden");
        showToast(`Buyer "${name}" saved across all devices.`);
      });

      document.getElementById("cancel-buyer-edit-btn").addEventListener("click", () => {
        document.getElementById("buyer-form").reset();
        document.getElementById("buyer-edit-doc-id").value = "";
        document.getElementById("buyer-form-heading").textContent = "Add New Buyer";
        document.getElementById("cancel-buyer-edit-btn").classList.add("hidden");
      });

      // Admin & Agent Credential Updates
      document.getElementById("admin-cred-form").addEventListener("submit", async (e) => {
        e.preventDefault();
        if (state.sessionRole !== "admin") return;
        const adminEmail = document.getElementById("settings-admin-email").value.trim();
        const adminPassword = document.getElementById("settings-admin-password").value;
        await saveCredentialsToStore({ adminEmail, adminPassword });
        showToast("Admin login credentials updated across all devices.");
      });

      document.getElementById("agent-cred-form").addEventListener("submit", async (e) => {
        e.preventDefault();
        if (state.sessionRole !== "admin") return;
        const agentEmail = document.getElementById("settings-agent-email").value.trim();
        const agentPassword = document.getElementById("settings-agent-password").value;
        await saveCredentialsToStore({ agentEmail, agentPassword });
        showToast("Agent login credentials updated across all devices.");
      });

      document.getElementById("export-csv-btn").addEventListener("click", exportLeadsCsv);

      document.getElementById("download-json-btn").addEventListener("click", () => {

          {
            exportedAt: new Date().toISOString(),
            buyers: state.buyers,
            leads: state.leads
          },
          null,
          2
        );
        const blob = new Blob([payload], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `hotline-bpo-backup-${new Date().toISOString().slice(0, 10)}.json`;
        document.body.appendChild(a);
        a.click();
        a.remove();
        URL.revokeObjectURL(url);
        showToast("JSON snapshot downloaded.");
      });
    }
